cmake_minimum_required(VERSION 3.20)
project(jqe CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")
option(JQE_NATIVE "Tune for the build machine" ON)
option(JQE_SANITIZE "Address/UB sanitizers" OFF)

add_library(jqe_core STATIC src/tables.cpp src/position.cpp src/eval.cpp src/nnue.cpp src/search.cpp)
target_include_directories(jqe_core PUBLIC src)
target_compile_options(jqe_core PRIVATE -Wall -Wextra)
find_package(Threads REQUIRED)
target_link_libraries(jqe_core PUBLIC Threads::Threads)
if(JQE_NATIVE)
    target_compile_options(jqe_core PUBLIC -march=native)
endif()
if(JQE_SANITIZE)
    target_compile_options(jqe_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(jqe_core PUBLIC -fsanitize=address,undefined)
endif()

add_executable(jqe src/main.cpp src/uci.cpp src/selfplay.cpp)
target_link_libraries(jqe PRIVATE jqe_core)
target_compile_options(jqe PRIVATE -Wall -Wextra)

# Differential test against the project's independent rules oracle (native/, original code).
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../native ${CMAKE_BINARY_DIR}/native_oracle EXCLUDE_FROM_ALL)
add_executable(jqe_difftest tests/difftest.cpp)
target_link_libraries(jqe_difftest PRIVATE jqe_core jieqi_core)
